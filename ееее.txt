POSTGRES_USER=django_user   
POSTGRES_PASSWORD=mysecretpassword1234
POSTGRES_DB=db_kittygram

DB_HOST=db_kittygram
DB_PORT=5432






# Подключаемся к контейнеру db_kittygram и запускаем psql с пользователем kittygram_user и базой данных kittygram
docker exec -it db_kittygram psql -U kittygram_user -d kittygram

# Удаляем все образы Docker, которые не используются (флаг -q возвращает только идентификаторы образов)
docker rmi $(docker images -q) 

# Создаем новый том с именем pgdata для хранения данных PostgreSQL
docker volume create pgdata

# Останавливаем контейнер db_kittygram
docker container stop db_kittygram

# Удаляем контейнер db_kittygram
docker container rm db_kittygram

# Удаляем сеть kittygram-network
docker network rm kittygram-network

# Создаем новую сеть с именем kittygram-network
docker network create kittygram-network

# Подключаем контейнер db_kittygram к сети kittygram-network
docker network connect kittygram-network db_kittygram

# Инспектируем сеть kittygram-network, чтобы увидеть подключенные контейнеры и их параметры
docker network inspect kittygram-network

# Запускаем новый контейнер db_kittygram с конфигурацией из файла .env и подключаем том pgdata для хранения данных PostgreSQL
docker run --name db_kittygram --env-file .env -v pgdata:/var/lib/postgresql/data postgres:13

# Запускаем контейнер kittygram_backend_container с конфигурацией из файла .env, подключаем его к сети kittygram-network и перенаправляем порт 9000 на 8000
docker run --env-file .env --net kittygram-network --name kittygram_backend_container -p 9000:8000 kittygram_backend

# Выполняем миграции базы данных в контейнере kittygram_backend_container
docker exec kittygram_backend_container python manage.py migrate 

# Подключаемся к контейнеру db_kittygram и запускаем psql с пользователем django_user и базой данных kittygram
docker exec -it db_kittygram psql -U django_user -d kittygram 

# Строим образ Docker с тегом kittygram_backend из текущего контекста (Dockerfile должен находиться в текущей директории)
docker build -t kittygram_backend .
